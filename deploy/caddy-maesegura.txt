# Bloco a acrescentar ao Caddyfile de /opt/wooneya-backend.
# Colocar junto dos restantes blocos, nunca num Caddy novo.
#
# Domínio oficial: maesegura.energisa.co.mz — a Energisa é a empresa candidata
# à chamada do UNICEF, e o protótipo tem de viver no domínio dela.
# O antigo maesegura.wooneya.co.mz fica só a reencaminhar, para que qualquer
# ligação já partilhada continue a funcionar.

maesegura.energisa.co.mz {
	encode zstd gzip
	reverse_proxy maesegura-web:8000
	header {
		Strict-Transport-Security "max-age=31536000; includeSubDomains"
		X-Content-Type-Options "nosniff"
		Referrer-Policy "same-origin"
		-Server
	}
	log {
		output file /var/log/caddy/maesegura.log
	}
}

maesegura.wooneya.co.mz {
	redir https://maesegura.energisa.co.mz{uri} permanent
}
